\section{Combinación PID + Predictor Smith + Dahlin.}
Primero se plantea

\begin{equation}
  G_p(z^{-1}) = \frac{bz^{-1}}{1-az^{-1}}z^{-N}
  \label{planta-ps}
\end{equation}
\begin{equation}
  G_m(z^{-1}) = \frac{bz^{-1}}{1-az^{-1}}
  \label{modelo-ps}
\end{equation}
\[
  \alpha = e^{-t_s/\tau}
\]
\[
  \beta = k(1 - \alpha)
\]
Luego por algebra de bloques se obtiene la función de transferencia del controlador con predictor smith.

\[
  u(z^{-1}) = G_c(z^{-1}) e_2(z^{-1}) = G_c(z^{-1})(e_1(z^{-1}) - \hat{b}(1 - z^{-N})))
\]

\[
  u(z^{-1}) = G_c(z^{-1})(e_1(z^{-1}) - u(z^{-1})G_m(z^{-1})(1 - z^{-N})))
\]

\[
  u(z^{-1})(1 + G_c(z^{-1})G_m(z^{-1})(1 - z^{-N}))) = G_c(z^{-1})(e_1(z^{-1}))
\]

Y así finalmente, la ecuación del controaldor del predictor smith es:
\begin{equation}
  \frac{u(z^{-1})}{e_1(z^{-1})} = \frac{G_c(z^{-1})}{1 + G_c(z^{-1})G_m(z^{-1}) -G_c(z^{-1})G_m(z^{-1})z^{-N}}
  \label{GC_ps}
\end{equation}

Ahora, tomando la estructura general de la función de transferencia en lazo cerrado se tiene:

\[
  H_{LC} = \frac{G_pG_c}{1 + G_pG_c} = \frac{\beta z^{-N}}{1 - \alpha z^{-1}}
\]

Luego, despejando el controldor de la ecuación general,

\[
  G_c = \frac{H_{LC}}{1 - H_{LC}}\cdot \frac{1}{G_p}
\]

Tomando las ecuaciones de la planta y lazo cerrado se reemplaza,

\begin{equation}
  G_c = \frac{\frac{\beta}{b}(1 - az^{-1})}{1 - \alpha z^{-1} - \beta z^{-1-N}}
  \label{gc_normal}
\end{equation}

Ahora igualando la ecuación del controlador general \ref{gc_normal} con la del controlador con predictor smith \ref{GC_ps}, es posible determinar los parámetros para diseñar un controlador PI tal que se comporte como un controlador con predictor smith.

\[
 G_c(z^{-1}) = \frac{\frac{\beta}{b}(1 - az^{-1})}{1 - \alpha z^{-1} - \beta z^{-1-N}} =\frac{G_c(z^{-1})}{1 + G_c(z^{-1})G_m(z^{-1}) -G_c(z^{-1})G_m(z^{-1})z^{-N}} 
\]

\[
  \frac{\frac{\beta}{b}(1 - az^{-1})}{1 - \alpha z^{-1} - \beta z^{-1-N}} =\frac{G_c(z^{-1})(a - z^{-1})}{(1 - az^{-1}) + G_c(z^{-1})(bz^{-1}) -G_c(z^{-1})bz^{-1-N}} 
\]

\[
  \frac{\beta}{b}(1-az^{-1} + G_c(z^{-1})b z^{-1} - G_c(z^{-1})(b z^{-(1+N)})) = G_c(z^{-1})(1 - \alpha z^{-1} - \beta z^{z^{-(1+N)}})
\]


\[
  \frac{\beta}{b}(1-az^{-1}) = G_c(z^{-1})(1 - \alpha z^{-1} - \beta z^{-(1+N)} - \beta z^{-1} +\beta z^{-(1+N)})
\]
Y finalmente,

\begin{equation}
  G_c = \frac{\frac{\beta}{b}(1 - az^{-1})}{1 - \alpha z^{-1} - \beta z^{-1}} 
  \label{controlador-ps}
\end{equation}

Ahora, despejando los componentes del controlador se llega a 

\[
  u(z^{-1})(1 - \alpha z^{-1} - \beta z^{-1}) = e(z^{-1})\frac{\beta}{b}(1 - az^{-1})
\]


\[
  u_k - u_{k-1}(\alpha - \beta) = e_k\frac{\beta}{b} - e_{k-1}a\frac{\beta}{b}
\]

\[
  u_k - u_{k-1}(\alpha - k(1-\alpha)) = e_k\frac{\beta}{b} - e_{k-1}a\frac{\beta}{b}
\]

Para que la salida sea igual al $SP$ deseado, la ganancia dc debe ser unitaria por lo que $K = 1$ y entonces.


\[
  u_k  = u_{k-1} + e_k\frac{\beta}{b} - e_{k-1}a\frac{\beta}{b}
\]

Luego, para encontrar los parámetros del \textbf{PI},

\[
  u_k  = u_{k-1} + e_k\frac{1 - e^{-\frac{t_s}{\tau}}}{b} - e_{k-1}a\frac{1 -e^{\frac{t_s}{\tau}}}{b}
\]

Comparando con los parámetros del PI clásico.

\[
  u_k = u_{k-1} + e_kk_c(1 + \frac{T_s}{2T_i}) + e_{k-1}k_c(-1 + \frac{T_s}{2T_i})
\]

\[
  \therefore k_c(1 + \frac{T_s}{2T_i}) = \frac{\beta}{b}, \implies k_c = \frac{2T_i \beta}{2T_ib +T_s}
\]

\[
  -\frac{\beta a}{b} = (\frac{2T_i \beta}{2T_i b + T_s})(-1 + \frac{1}{2T_i})
\]


\[
  -\frac{\beta a}{b} = (\frac{\beta - 2T_i \beta}{2T_i b + T_s})
\]

\[
  -\frac{\beta a}{b}(2T_i b + T_s) = \beta - 2T_i \beta
\]


\[
  2T_i(\beta - \beta a) = \beta + \beta T_s
\]


\[
  T_i = \frac{1 + T_s}{2(1 - a)}
\]
